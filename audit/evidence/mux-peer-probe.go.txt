package kernel
import("net";"strconv";"testing";"io";"time")
func TestUpstreamMuxChangesKernelPeer(t *testing.T){
 backend,err:=net.Listen("tcp","127.0.0.1:0");if err!=nil{t.Fatal(err)};defer backend.Close()
 seen:=make(chan string,1)
 go func(){c,e:=backend.Accept();if e!=nil{return};defer c.Close();seen<-c.RemoteAddr().String();b:=make([]byte,4);io.ReadFull(c,b);c.Write(b)}()
 mux:=newSNIMux();defer mux.stop();port:=freeLocalTCPPort(t)
 if _,err:=mux.apply(MuxConfig{Enabled:true,Listen:"127.0.0.1",Port:port,FallbackTarget:backend.Addr().String()});err!=nil{t.Fatal(err)}
 dial:=net.Dialer{LocalAddr:&net.TCPAddr{IP:net.ParseIP("127.0.0.2")}}
 c,err:=dial.Dial("tcp",net.JoinHostPort("127.0.0.1",strconv.Itoa(port)));if err!=nil{t.Fatal(err)};defer c.Close();c.Write([]byte("GET "))
 select{case peer:=<-seen:h,_,_:=net.SplitHostPort(peer);if h!="127.0.0.1"{t.Fatalf("peer=%s",h)};t.Log("client source 127.0.0.2 became kernel peer 127.0.0.1; native inbound route does not send PROXY")
 case <-time.After(time.Second):t.Fatal("no backend connection")}
}
